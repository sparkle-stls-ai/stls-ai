# local
#########
redis-cli

KEYS *
KEYS stls:pg:*

# remaining TTL of a key (in seconds)
TTL stls:ks:default:key:<kid>

# check that it exists (1 = exists, 0 = doesn't exist / evicted)
EXISTS stls:ks:default:key:<kid>

# length in bytes (e.g. 32 for AES-256)
STRLEN stls:ks:default:key:<kid>

# do NOT GET a cryptographic key in production — dev/debug only
GET stls:ks:default:key:<kid>

# all pending kids in the queue, with insertion timestamp (score)
ZRANGE stls:ks:default:pending 0 -1 WITHSCORES

# how many keys are pending
ZCARD stls:ks:default:pending

# pending keys that arrived after a given timestamp
ZRANGEBYSCORE stls:ks:default:pending 1750000000 +inf WITHSCORES

ZCARD stls:ks:default:pending          # → 1
TTL  stls:ks:default:key:<kid>         # → ~300 (KEY_STORE_PENDING_TTL)


# remove a specific key
DEL stls:ks:default:key:<kid>
ZREM stls:ks:default:pending <kid>

# clear the whole namespace (WARNING: dev only)
EVAL "return redis.call('DEL', unpack(redis.call('KEYS', ARGV[1])))" 0 "stls:ks:default:*"

# redis monitoring
redis-cli -u redis://localhost:6379 MONITOR | grep "stls:ks"
